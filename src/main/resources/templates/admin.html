Relational Database Schema Architecture

stations Table

Stores all operational Mumbai & Navi Mumbai Metro stations across all 7 lines (Lines 1, 2A, 7, 9, 3, 2B, NMM-1) with distance markers and interchange links. 

CREATE TABLE stations (
    id INTEGER PRIMARY KEY,
    name TEXT NOT NULL,
    line_code TEXT NOT NULL,
    line_name TEXT NOT NULL,
    distance_km REAL NOT NULL,
    interchange TEXT DEFAULT 'None',
    UNIQUE(line_code, name)
);

cards Table

Stores issued smart cards with polymorphic type attributes, balance, and account security flags. 

CREATE TABLE cards (
    card_id TEXT PRIMARY KEY,
    holder_name TEXT NOT NULL,
    contact_number TEXT,
    card_type TEXT NOT NULL,
    balance REAL NOT NULL,
    is_blocked INTEGER NOT NULL DEFAULT 0,
    created_at TEXT NOT NULL
);

tickets Table

Stores paper & QR digital tokens linked to origin and destination station Foreign Keys. 

CREATE TABLE tickets (
    ticket_id TEXT PRIMARY KEY,
    source_id INTEGER NOT NULL,
    dest_id INTEGER NOT NULL,
    distance_km REAL NOT NULL,
    unit_fare REAL NOT NULL,
    passenger_count INTEGER DEFAULT 1,
    is_return INTEGER DEFAULT 0,
    total_fare REAL NOT NULL,
    ticket_type TEXT NOT NULL,
    qr_data TEXT,
    booked_at TEXT NOT NULL,
    FOREIGN KEY (source_id) REFERENCES stations(id),
    FOREIGN KEY (dest_id) REFERENCES stations(id)
);

transactions Table

Immutable financial and journey audit log tracking gate tap-ins, tap-outs, deductions, and top-ups. 

CREATE TABLE transactions (
    txn_id TEXT PRIMARY KEY,
    ref_id TEXT NOT NULL,
    txn_type TEXT NOT NULL,
    amount REAL NOT NULL,
    source_station TEXT,
    dest_station TEXT,
    balance_after REAL DEFAULT 0.0,
    status TEXT NOT NULL,
    created_at TEXT NOT NULL
);
